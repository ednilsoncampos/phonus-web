# 28 - Cenário: ciclo de vida dos usuários de uma empresa (cadastro → convites → gestão)
# Dependências: banco zerado e SUPER_ROOT dev desligado (padrão: SUPER_ROOT_HABILITADO=false); os
#   e-mails abaixo são únicos na plataforma. Os tokens de ativação/convite são lidos no banco, então
#   não é preciso abrir a caixa de e-mail (a Brevo envia os e-mails de verdade).
# Versão executável: docs/homologacao/scripts/cenario-ciclo-usuarios.sh (79 verificações)
#
# Validado em 2026-10-01: 79 ok, 0 falha.
#
# E-mails (caixas reais do projeto):
#   ROOT        ednilsoncampos@gmail.com            (cadastra a empresa)
#   ADMIN       ednilson.campos.dev@gmail.com       (convidado pelo ROOT)
#   OPERADOR    maria.cs.2239@gmail.com             (convidado pelo ROOT)
#   OPERADOR 2  camposolution.suporte@gmail.com     (convidado pelo ROOT; usado em desativar/reativar)
#
# Cenário:
#   1. Cadastrar a empresa e o ROOT e ativar a conta pelo link
#   2. O ROOT convida ADMIN, OPERADOR e um 2º OPERADOR; cada um define a senha pelo link do convite
#   3. Identidade de cada papel (/auth/me, GET /usuarios/{id}, entitlement)
#   4. O ROOT gerencia usuários: promover/rebaixar, desativar/reativar, reenviar convite
#   5. Recusas da hierarquia
#   6. Escopo financeiro com os usuários criados
#   7. Troca de senha invalida os tokens anteriores
#
# Variáveis: {{termosId}} (GET /termos/atual), {{empresaId}}, {{accessTokenRoot}}, {{accessTokenAdmin}},
# {{accessTokenOperador}}, {{accessTokenOperador2}}, {{idRoot}}, {{idAdmin}}, {{idOperador}}, {{idOperador2}}
#
# Tokens de ativação/convite sem abrir o e-mail (schema = t_<empresaId sem hífens>):
#   SELECT t.token FROM "t_<...>".token_ativacao_usuario t JOIN "t_<...>".usuario u ON u.id = t.usuario_id
#   WHERE u.email = '<email>' AND NOT t.usado AND t.substituido_em IS NULL ORDER BY t.criado_em DESC LIMIT 1;
# O empresaId sai de: SELECT empresa_id FROM public.usuario_global WHERE email = '<email do ROOT>';
# Senhas: ROOT A.123456a; ADMIN, OPERADOR e OPERADOR 2 definem B.123456b no convite; ao final do passo 7 o ADMIN
#   passa a C.123456c.
# Atenção: o link de ativação do CADASTRO vale 3 minutos; o do CONVITE, 24 horas.


# =========================================================================
# 28.1 - Passo 1: cadastrar a empresa e o ROOT
# =========================================================================

curl --location 'http://localhost:8080/api/v1/auth/registro' \
--header 'Content-Type: application/json' \
--data-raw '{
    "nomeEmpresa": "Empresa Ciclo de Usuarios",
    "tipoDocumento": "CNPJ",
    "documento": "12.345.678/0001-95",
    "nome": "Ednilson Root",
    "email": "ednilsoncampos@gmail.com",
    "senha": "A.123456a",
    "termosId": "{{termosId}}"
}'

# Resposta esperada: 201, papel ROOT, status CONVIDADO (a conta só vale depois de ativada).

# --- Login antes de ativar (esperado 403: senha certa, conta não ativada) ---
curl --location 'http://localhost:8080/api/v1/auth/login' \
--header 'Content-Type: application/json' \
--data-raw '{ "email": "ednilsoncampos@gmail.com", "senha": "A.123456a" }'

# --- Ativar pelo link do e-mail (GET; o token vem do banco) → 200 "Conta Ativada!" ---
curl --location 'http://localhost:8080/api/v1/auth/ativar?token=TOKEN_DO_CADASTRO&e={{empresaId}}'

# --- O mesmo link de novo (esperado 400: token já usado) ---
curl --location 'http://localhost:8080/api/v1/auth/ativar?token=TOKEN_DO_CADASTRO&e={{empresaId}}'

# --- Login → {{accessTokenRoot}} (esperado 200); /auth/me traz papel ROOT e status ATIVO ---
curl --location 'http://localhost:8080/api/v1/auth/login' \
--header 'Content-Type: application/json' \
--data-raw '{ "email": "ednilsoncampos@gmail.com", "senha": "A.123456a" }'


# =========================================================================
# 28.2 - Passo 2: o ROOT convida os outros papéis
# =========================================================================

curl --location 'http://localhost:8080/api/v1/usuarios' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer {{accessTokenRoot}}' \
--data-raw '{ "nome": "Admin Dev", "email": "ednilson.campos.dev@gmail.com", "papel": "ADMIN" }'

curl --location 'http://localhost:8080/api/v1/usuarios' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer {{accessTokenRoot}}' \
--data-raw '{ "nome": "Maria Operadora", "email": "maria.cs.2239@gmail.com", "papel": "OPERADOR" }'

curl --location 'http://localhost:8080/api/v1/usuarios' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer {{accessTokenRoot}}' \
--data-raw '{ "nome": "Suporte Operador", "email": "camposolution.suporte@gmail.com", "papel": "OPERADOR" }'

# Resposta esperada: 201 nos três (status CONVIDADO). Enquanto não ativam, o login deles dá 401
# (a senha é desconhecida; o 403 "conta não ativada" só aparece com a senha certa).
# GET /usuarios (ROOT) lista 3 CONVIDADO + 1 ATIVO.

# --- Reenvio do convite ---
# Dentro de 2 minutos do último e-mail é ignorado (cooldown), com a mesma resposta 200 de sempre.
# Fora do cooldown gera um link novo e o antigo passa a dar 400.
curl --location 'http://localhost:8080/api/v1/auth/reenviar-ativacao' \
--header 'Content-Type: application/json' \
--data-raw '{ "email": "camposolution.suporte@gmail.com" }'

# Recusas (28.5) com e-mails que já existem: o papel é validado antes do e-mail, então nada é enviado.

# --- Cada convidado define a senha pelo formulário do convite (esperado 200 "Conta Ativada!") ---
curl --location 'http://localhost:8080/api/v1/auth/ativar' \
--header 'Content-Type: application/x-www-form-urlencoded' \
--data-urlencode 'token=TOKEN_DO_CONVITE' \
--data-urlencode 'e={{empresaId}}' \
--data-urlencode 'novaSenha=B.123456b' \
--data-urlencode 'confirmaSenha=B.123456b'

# Senha fraca no formulário (ex.: "somenteletras") → 400 e o convite continua valendo. Depois logar (04):
# {{accessTokenAdmin}}, {{accessTokenOperador}}, {{accessTokenOperador2}} (senha B.123456b)


# =========================================================================
# 28.3 - Passo 3: identidade de cada papel
# =========================================================================

# Trocar o token e o id conforme o papel. Esperado: 200 nos três papéis.
curl --location 'http://localhost:8080/api/v1/auth/me' \
--header 'Authorization: Bearer {{accessTokenAdmin}}'

curl --location 'http://localhost:8080/api/v1/usuarios/{{idAdmin}}' \
--header 'Authorization: Bearer {{accessTokenAdmin}}'

# Entitlement do próprio usuário: tier PREMIUM (cortesia de 21 dias), planoAtual null
curl --location 'http://localhost:8080/api/v1/usuarios/{{idAdmin}}/entitlement' \
--header 'Authorization: Bearer {{accessTokenAdmin}}'

# Cada usuário só consulta os próprios dados (esperado 403, inclusive para o ROOT):
#   GET /usuarios/{{idAdmin}}              com o token do ROOT
#   GET /usuarios/{{idRoot}}/entitlement   com o token do OPERADOR
#
# /auth/me → permissoes[]: USUARIOS_ALTERAR_PAPEL só aparece para o ROOT.
# GET /usuarios: ROOT e ADMIN 200 (4 usuários), OPERADOR 403.


# =========================================================================
# 28.4 - Passo 4: o ROOT gerencia usuários
# =========================================================================

# --- Promover o OPERADOR a ADMIN (esperado 200) e conferir que o token dele já vale como ADMIN ---
curl --location --request PATCH 'http://localhost:8080/api/v1/usuarios/{{idOperador}}/papel' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer {{accessTokenRoot}}' \
--data-raw '{ "papel": "ADMIN" }'

curl --location 'http://localhost:8080/api/v1/usuarios' \
--header 'Authorization: Bearer {{accessTokenOperador}}'
# Esperado 200 (o papel é lido do banco a cada requisição; não precisa logar de novo)

# --- Rebaixar de volta a OPERADOR (esperado 200); a mesma listagem volta a dar 403 ---
curl --location --request PATCH 'http://localhost:8080/api/v1/usuarios/{{idOperador}}/papel' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer {{accessTokenRoot}}' \
--data-raw '{ "papel": "OPERADOR" }'

# --- Desativar o 2º OPERADOR (esperado 204) ---
curl --location --request DELETE 'http://localhost:8080/api/v1/usuarios/{{idOperador2}}' \
--header 'Authorization: Bearer {{accessTokenRoot}}'

# Depois de desativado:
#   - o banco marca status INATIVO e a linha sai de public.usuario_global
#   - login → 401; o token antigo → 401 em qualquer rota
#   - POST /auth/reenviar-ativacao responde 200 mas NÃO o revive (só convite pendente é reenviado)

# --- Reativar (esperado 204): volta a logar com a mesma senha ---
curl --location --request PATCH 'http://localhost:8080/api/v1/usuarios/{{idOperador2}}/reativar' \
--header 'Authorization: Bearer {{accessTokenRoot}}'

# --- O ADMIN também desativa e reativa um OPERADOR (esperado 204 nos dois) ---
curl --location --request DELETE 'http://localhost:8080/api/v1/usuarios/{{idOperador}}' \
--header 'Authorization: Bearer {{accessTokenAdmin}}'

curl --location --request PATCH 'http://localhost:8080/api/v1/usuarios/{{idOperador}}/reativar' \
--header 'Authorization: Bearer {{accessTokenAdmin}}'
# Logar de novo para renovar {{accessTokenOperador}} e {{accessTokenOperador2}}


# =========================================================================
# 28.5 - Passo 5: recusas da hierarquia
# =========================================================================
#
# Chamada                                              | Quem       | Esperado
# -----------------------------------------------------|------------|---------
# DELETE /usuarios/{{idRoot}}                          | ADMIN      | 403
# PATCH  /usuarios/{{idOperador}}/papel                | ADMIN      | 403
# POST   /usuarios (papel ADMIN)                       | ADMIN      | 403 "ADMIN só pode convidar usuários com papel inferior ao seu"
# POST   /usuarios (papel ROOT)                        | ADMIN      | 403
# POST   /usuarios (qualquer papel)                    | OPERADOR   | 403
# PATCH  /usuarios/{{idRoot}}/papel                    | ROOT       | 403 "Não é permitido alterar o próprio papel"
# DELETE /usuarios/{{idRoot}}                          | ROOT       | 403 "Não é permitido desativar a própria conta"
# PATCH  /usuarios/{{idOperador}}/papel {"papel":"ROOT"} | ROOT     | 403 "O papel ROOT só pode ser atribuído durante o onboarding da empresa"
# POST   /usuarios (e-mail já cadastrado)              | ROOT       | 409
# PATCH  /usuarios/{{idOperador}}/reativar (já ativo)  | ROOT       | 422 "Usuário já está ativo"
# PATCH  /usuarios/00000000-0000-0000-0000-000000000000/papel | ROOT | 404

curl --location --request PATCH 'http://localhost:8080/api/v1/usuarios/{{idRoot}}/papel' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer {{accessTokenRoot}}' \
--data-raw '{ "papel": "ADMIN" }'

# Corpo de erro (todos): { "timestamp", "status", "error", "message", "path", "details" }


# =========================================================================
# 28.6 - Passo 6: escopo financeiro
# =========================================================================

# ADMIN, OPERADOR e OPERADOR 2 lançam uma venda cada (esperado 201); trocar o token e a descrição:
curl --location 'http://localhost:8080/api/v1/lancamentos' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer {{accessTokenAdmin}}' \
--data-raw '{
    "tipo": "ENTRADA_CAIXA",
    "descricao": "Venda do admin",
    "valorTotal": 5000,
    "formaPagamento": "PIX",
    "origem": "TEXTO",
    "dataLancamento": "2026-10-01"
}'
# (operador: "Venda do operador", 2000; operador 2: "Venda do operador 2", 1000)

# Escopo: ROOT e ADMIN veem os 3 lançamentos; cada OPERADOR vê só o próprio
curl --location 'http://localhost:8080/api/v1/lancamentos' \
--header 'Authorization: Bearer {{accessTokenOperador}}'

# --- ROOT lê o lançamento de um operador (200); o OPERADOR lê o do ADMIN (403) ---
curl --location 'http://localhost:8080/api/v1/lancamentos/UUID_DO_LANCAMENTO_DO_ADMIN' \
--header 'Authorization: Bearer {{accessTokenOperador}}'

# --- Caixa do ROOT soma a empresa inteira: saldoEmCentavos 8000 ---
curl --location 'http://localhost:8080/api/v1/caixa' \
--header 'Authorization: Bearer {{accessTokenRoot}}'


# =========================================================================
# 28.7 - Passo 7: troca de senha invalida os tokens anteriores
# =========================================================================

curl --location --request PUT 'http://localhost:8080/api/v1/auth/senha' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer {{accessTokenAdmin}}' \
--data-raw '{ "senhaAtual": "B.123456b", "novaSenha": "C.123456c" }'

# Resposta esperada: 204. Depois:
#   - o token antigo do ADMIN → 401 em qualquer rota (credenciais_alteradas_em)
#   - login com a senha antiga → 401; com a nova (C.123456c) → 200
#   - nova troca informando a senha atual errada → 422 "Senha atual incorreta"
